"class" "" "eager" false "widths" (slice 480 960 1600)) */ -}} {{- $img := .img -}} {{- $widths := .widths | default (slice 480 800 1200 1600) -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q78" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- $fallback = $r -}} {{- end -}} {{- end -}} {{- if not $fallback }}{{ $fallback = $img.Resize (printf "%dx webp q78" $img.Width) }}{{ $srcset = slice (printf "%s %dw" $fallback.RelPermalink $img.Width) }}{{ end -}} {{- /* */ -}}
